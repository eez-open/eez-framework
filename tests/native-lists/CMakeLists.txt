cmake_minimum_required(VERSION 3.16)
project(eez_native_lists C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(EEZ_TEST_SANITIZE "Enable AddressSanitizer and UndefinedBehaviorSanitizer" ON)
file(GLOB_RECURSE SOURCES "../../src/eez/*.cpp" "../../src/eez/*.c")
list(FILTER SOURCES EXCLUDE REGEX "/libs/libscpi/")
add_executable(eez-native-lists test.cpp ${SOURCES})
target_include_directories(eez-native-lists PRIVATE . ../../src)
if(EEZ_COMPILER_FIXTURE_DIR)
    target_include_directories(eez-native-lists PRIVATE "${EEZ_COMPILER_FIXTURE_DIR}")
    target_compile_definitions(eez-native-lists PRIVATE EEZ_HAS_COMPILER_FIXTURE)
endif()
if(EEZ_TEST_SANITIZE)
    target_compile_options(eez-native-lists PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(eez-native-lists PRIVATE -fsanitize=address,undefined)
endif()
